// 1. Menerima data kiriman dari Ubuntu / PHP (POST) function doPost(e) { try { var contents = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); if (contents.type === 'ap_distribution') { var sheet = ss.getSheetByName("AP_Stats"); if (!sheet) { sheet = ss.insertSheet("AP_Stats"); sheet.appendRow(["Timestamp", "APName", "ClientCount"]); } // Clear data lama var lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.getRange(2, 1, lastRow - 1, 3).clearContent(); } var timestamp = new Date(); var rows = []; contents.data.forEach(function(item) { rows.push([timestamp, item.APName, item.ClientCount]); }); if (rows.length > 0) { sheet.getRange(2, 1, rows.length, 3).setValues(rows); } return ContentService.createTextOutput(JSON.stringify({ status: "success", rows_inserted: rows.length })) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Unknown payload type" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // 2. Mengirim data ke Dashboard HTML di Vercel (GET) function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); // Read AP_Stats Sheet var apSheet = ss.getSheetByName("AP_Stats"); var apData = []; if (apSheet && apSheet.getLastRow() > 1) { var values = apSheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][1]) { apData.push({ APName: values[i][1], ClientCount: values[i][2] }); } } } // Read Devices / Inventory (jika ada sheet Devices) var devSheet = ss.getSheetByName("Devices"); var devData = []; if (devSheet && devSheet.getLastRow() > 1) { var dValues = devSheet.getDataRange().getValues(); for (var j = 1; j < dValues.length; j++) { devData.push({ DeviceName: dValues[j][0] || "CCR2004", ActiveClients: dValues[j][1] || 0, CpuLoad: dValues[j][2] || "0%" }); } } else { // Default fallback jika sheet Devices belum diisi var totalClients = apData.reduce(function(sum, item) { return sum + Number(item.ClientCount || 0); }, 0); devData.push({ DeviceName: "CCR2004-16G-2S+", ActiveClients: totalClients, CpuLoad: "5%" }); } // Response JSON sesuai ekspektasi index.html var response = { status: "success", devices: devData, wans: [], // Akan terisi jika Anda menambah sheet WAN_Stats vlans: [], // Akan terisi jika Anda menambah sheet VLAN_Stats ap_distribution: apData }; return ContentService.createTextOutput(JSON.stringify(response)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
